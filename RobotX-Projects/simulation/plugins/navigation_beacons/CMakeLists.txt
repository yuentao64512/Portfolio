cmake_minimum_required(VERSION 3.16)
project(NavigationBeacons LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
add_library(NavigationBeacons SHARED NavigationBeacons.cc)
target_compile_features(NavigationBeacons PRIVATE cxx_std_17)
target_link_libraries(NavigationBeacons PRIVATE gz-sim8::gz-sim8)
set_target_properties(NavigationBeacons PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../lib")
include(CTest)
if(BUILD_TESTING)
  add_executable(beacon_timing_test beacon_timing_test.cc)
  target_compile_features(beacon_timing_test PRIVATE cxx_std_17)
  target_link_libraries(beacon_timing_test PRIVATE gz-sim8::gz-sim8)
  add_test(NAME beacon_timing COMMAND beacon_timing_test)
endif()
